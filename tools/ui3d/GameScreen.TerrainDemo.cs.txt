    frame=ScreenCapture.CaptureScreenshotAsTexture();
    string fmt=frame.format==TextureFormat.RGB24?"rgb24":frame.format==TextureFormat.RGBA32?"rgba":"";
    if(fmt=="")throw new Exception("Unsupported screenshot format: "+frame.format);
    if(demoEncoder==null){var info=new ProcessStartInfo(demoFfmpeg,"-hide_banner -loglevel error -y -f rawvideo -pixel_format "+fmt+" -video_size "+frame.width+"x"+frame.height+" -framerate 30 -i pipe:0 -vf vflip -c:v libx264 -preset ultrafast -crf 19 -pix_fmt yuv420p -movflags +faststart \""+demoFile+"\""){UseShellExecute=false,RedirectStandardInput=true,RedirectStandardError=true,CreateNoWindow=true,WindowStyle=ProcessWindowStyle.Hidden};demoEncoder=Process.Start(info);demoEncoder!.ErrorDataReceived+=(_,e)=>{if(!string.IsNullOrWhiteSpace(e.Data))File.AppendAllText(Path.Combine(demoFolder,"encoder.log"),e.Data+"\n");};demoEncoder.BeginErrorReadLine();}
    var bytes=frame.GetRawTextureData();int goal=Mathf.FloorToInt((Time.realtimeSinceStartup-demoStarted)*30)+1;
    while(demoFrames<goal){demoEncoder!.StandardInput.BaseStream.Write(bytes,0,bytes.Length);demoFrames++;}
   }catch(Exception e){demoError=e.ToString();demoRecording=false;}finally{if(frame!=null)UnityEngine.Object.Destroy(frame);}}}

  private void DemoMove(VisualElement target,float x=.5f,float y=.5f){demoPointer=new Vector2(Mathf.Lerp(target.worldBound.xMin,target.worldBound.xMax,x),Mathf.Lerp(target.worldBound.yMin,target.worldBound.yMax,y));using(var e=PointerMoveEvent.GetPooled(new Event{type=EventType.MouseMove,mousePosition=demoPointer})){e.target=target;target.SendEvent(e);}}

   board3DViewport.Focus=to;board3DViewport.Zoom=zoom;yield return new WaitForSecondsRealtime(.5f);
  }
